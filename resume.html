function downloadHighQualityPDF() { // Hide controls const controls = document.getElementById('controls'); const originalDisplay = controls.style.display; try { controls.style.display = 'none'; // Set up jsPDF and html2pdf parameters const pdfOptions = { margin: 0, filename: 'output.pdf', html2canvas: { scale: 2 }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'portrait', compressPDF: true } }; // Generate the PDF html2pdf().from(document.body).set(pdfOptions).save(); } catch (error) { console.error('Error generating PDF:', error); } finally { // Restore controls and settings controls.style.display = originalDisplay; optimizeForSinglePage(false); } }
